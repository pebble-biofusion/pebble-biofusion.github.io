<script>
  // Lightbox functionality
  const lightbox = document.getElementById('lightbox');
  const lightboxImage = document.getElementById('lightbox-image');
  const lightboxTitle = document.getElementById('lightbox-title');
  const lightboxDescription = document.getElementById('lightbox-description');

  // Open lightbox when clicking view button
  document.querySelectorAll('.view-btn').forEach(function(btn) {
    btn.addEventListener('click', function(e) {
      e.stopPropagation();
      const imgWrapper = this.closest('.gallery-image-wrapper');
      const img = imgWrapper.querySelector('.gallery-image');
      const fullSrc = img.dataset.full || img.src;
      const title = img.dataset.title || 'Workshop Photo';
      const description = img.dataset.description || '';

      lightboxImage.src = fullSrc;
      lightboxImage.alt = title;
      lightboxTitle.textContent = title;
      lightboxDescription.textContent = description;
      lightbox.classList.add('active');
    });
  });

  // Close lightbox
  document.querySelector('.lightbox-close').addEventListener('click', function() {
    lightbox.classList.remove('active');
  });

  // Close when clicking outside
  lightbox.addEventListener('click', function(e) {
    if (e.target === lightbox) {
      lightbox.classList.remove('active');
    }
  });

  // Download functionality
  document.querySelectorAll('.download-btn, .lightbox-download').forEach(function(btn) {
    btn.addEventListener('click', function(e) {
      e.stopPropagation();
      let imgSrc;
      
      if (this.classList.contains('lightbox-download')) {
        imgSrc = lightboxImage.src;
      } else {
        const imgWrapper = this.closest('.gallery-image-wrapper');
        const img = imgWrapper.querySelector('.gallery-image');
        imgSrc = img.dataset.full || img.src;
      }

      // Create download link
      const link = document.createElement('a');
      link.href = imgSrc;
      link.download = imgSrc.split('/').pop();
      link.click();
    });
  });

  // Share functionality
  document.querySelectorAll('.share-btn, .lightbox-share').forEach(function(btn) {
    btn.addEventListener('click', function(e) {
      e.stopPropagation();
      let shareUrl;
      
      if (this.classList.contains('lightbox-share')) {
        shareUrl = window.location.href.split('#')[0];
      } else {
        const imgWrapper = this.closest('.gallery-image-wrapper');
        const img = imgWrapper.querySelector('.gallery-image');
        shareUrl = img.dataset.full || img.src;
      }

      if (navigator.share) {
        navigator.share({
          title: 'Pebble BioFusion Workshop Photo',
          url: shareUrl
        });
      } else {
        // Fallback: copy to clipboard
        navigator.clipboard.writeText(shareUrl).then(function() {
          alert('Link copied to clipboard!');
        });
      }
    });
  });

  // Close lightbox with Escape key
  document.addEventListener('keydown', function(e) {
    if (e.key === 'Escape' && lightbox.classList.contains('active')) {
      lightbox.classList.remove('active');
    }
  });
</script>
